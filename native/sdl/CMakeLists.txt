cmake_minimum_required(VERSION 3.24)
project(julia_sdl LANGUAGES C)
include(FetchContent)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
if(FETCHCONTENT_SOURCE_DIR_SDL)
    message(FATAL_ERROR "Julia patches its own pristine SDL copy; unset FETCHCONTENT_SOURCE_DIR_SDL")
endif()
# A missing SOURCE_SUBDIR downloads the pinned archive without adding it; Julia builds a patched copy instead.
FetchContent_Declare(SDL
    URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.4.12
    URL_HASH SHA256=b68381f06a7580e63400b3b6eb547ec57d8c3ebde70f9f40e0aba530ba05da27
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    SOURCE_SUBDIR julia-download-only)
FetchContent_MakeAvailable(SDL)

# Patches always start from the pristine archive, so editing one never leaves a stale tree behind. The stamp
# skips the copy when nothing changed, which keeps SDL's own build incremental.
set(sdl_patched "${CMAKE_BINARY_DIR}/sdl-patched")
file(GLOB_RECURSE patch_inputs RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/common/*.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/metal/*.cmake"
    "${CMAKE_CURRENT_SOURCE_DIR}/vulkan/*.cmake")
list(SORT patch_inputs)
list(PREPEND patch_inputs CMakeLists.txt)
set(patch_identity "")
foreach(input IN LISTS patch_inputs)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${input}" input_hash)
    string(APPEND patch_identity "${input_hash}")
endforeach()
string(SHA256 patch_identity "${patch_identity}")
set(patch_stamp "${sdl_patched}/.julia-patches")
set(patched_identity "")
if(EXISTS "${patch_stamp}")
    file(READ "${patch_stamp}" patched_identity)
endif()
if(NOT patched_identity STREQUAL patch_identity)
    file(REMOVE_RECURSE "${sdl_patched}")
    file(COPY "${sdl_SOURCE_DIR}/" DESTINATION "${sdl_patched}")
    include("${CMAKE_CURRENT_SOURCE_DIR}/common/patching.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/common/exports.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/common/sampler_capacity.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/common/sparse_targets.cmake")
    # Each GPU backend folder patches only its own SDL driver.
    if(APPLE)
        set(backend "${sdl_patched}/src/gpu/metal/SDL_gpu_metal.m")
        include("${CMAKE_CURRENT_SOURCE_DIR}/metal/backend.cmake")
        include("${CMAKE_CURRENT_SOURCE_DIR}/metal/sampler_arguments.cmake")
        include("${CMAKE_CURRENT_SOURCE_DIR}/metal/sparse_targets.cmake")
        include("${CMAKE_CURRENT_SOURCE_DIR}/metal/visibility.cmake")
        file(APPEND "${backend}" "\n#include \"gpu_extensions.m\"\n")
    endif()
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/draw_parameters.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/vertex_input.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/sampler_arrays.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/buffer_storage.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/texture_views.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/sparse_targets.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/queries.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/buffer_textures.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/image_layouts.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/integer_samplers.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/vulkan/nonuniform_samplers.cmake")
    file(APPEND "${sdl_patched}/src/gpu/vulkan/SDL_gpu_vulkan.c" "\n#include \"gpu_extensions.c\"\n")
    file(WRITE "${patch_stamp}" "${patch_identity}")
endif()

add_subdirectory("${sdl_patched}" "${CMAKE_BINARY_DIR}/sdl-build")
# The backend sources include their extension files by name, compiled inside SDL's own translation unit.
if(APPLE)
    target_include_directories(SDL3-shared PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/metal")
endif()
target_include_directories(SDL3-shared PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vulkan")
target_sources(SDL3-shared PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/common/gpu_extensions.c")
set_target_properties(SDL3-shared PROPERTIES OUTPUT_NAME SDL3Julia)
