cmake_minimum_required(VERSION 3.24)
project(julia_sdl LANGUAGES C)
if(NOT APPLE)
    message(FATAL_ERROR "The native timestamp extension currently implements Metal on macOS only")
endif()
include(FetchContent)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL
    URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.4.12
    URL_HASH SHA256=b68381f06a7580e63400b3b6eb547ec57d8c3ebde70f9f40e0aba530ba05da27
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL)
# Compile inside SDL's backend so private Metal layouts never leak into Julia's Java ABI.
set(backend "${sdl_SOURCE_DIR}/src/gpu/metal/SDL_gpu_metal.m")
file(READ "${backend}" source)
string(FIND "${source}" "#include \"gpu_timestamps.m\"" patched)
if(patched EQUAL -1)
    file(APPEND "${backend}" "\n/* Julia extension: altered SDL source. */\n#include \"gpu_timestamps.m\"\n")
endif()
# Pin physical buffer ownership to the texture view until SDL retires its queued draws.
file(READ "${backend}" source)
string(FIND "${source}" "juliaBufferOwner" buffer_patched)
if(buffer_patched EQUAL -1)
    string(REPLACE "typedef struct MetalTexture\n{\n    id<MTLTexture> handle;"
        "typedef struct MetalTexture\n{\n    void *juliaBufferOwner; /* Julia extension: retained buffer storage. */\n    id<MTLTexture> handle;" source "${source}")
    string(REPLACE "        container->textures[i]->handle = nil;"
        "        if (container->textures[i]->juliaBufferOwner) {\n            SDL_AddAtomicInt(&((MetalBuffer *)container->textures[i]->juliaBufferOwner)->referenceCount, -1);\n        }\n        container->textures[i]->handle = nil;" source "${source}")
    file(WRITE "${backend}" "${source}")
endif()
file(READ "${backend}" source)
string(FIND "${source}" "juliaTextureReferences" storage_patched)
if(storage_patched EQUAL -1)
    string(REPLACE "typedef struct MetalBuffer\n{\n    id<MTLBuffer> handle;"
        "typedef struct MetalBuffer\n{\n    SDL_AtomicInt juliaTextureReferences; /* Julia extension: shared texture storage. */\n    id<MTLBuffer> handle;" source "${source}")
    string(REPLACE "SDL_AddAtomicInt(&((MetalBuffer *)container->textures[i]->juliaBufferOwner)->referenceCount, -1);"
        "SDL_AddAtomicInt(&((MetalBuffer *)container->textures[i]->juliaBufferOwner)->referenceCount, -1);\n            SDL_AddAtomicInt(&((MetalBuffer *)container->textures[i]->juliaBufferOwner)->juliaTextureReferences, -1);" source "${source}")
    string(REPLACE "static MetalBuffer *METAL_INTERNAL_PrepareBufferForWrite(\n    MetalRenderer *renderer,\n    MetalBufferContainer *container,\n    bool cycle)\n{"
        "static MetalBuffer *METAL_INTERNAL_PrepareBufferForWrite(\n    MetalRenderer *renderer,\n    MetalBufferContainer *container,\n    bool cycle)\n{\n    /* Julia extension: GL texture buffers share updates to the same physical store. */\n    if (SDL_GetAtomicInt(&container->activeBuffer->juliaTextureReferences) > 0) cycle = false;" source "${source}")
    file(WRITE "${backend}" "${source}")
endif()
# Texture retirement drops retained buffer references. Retire textures first, including on final device shutdown.
file(READ "${backend}" source)
string(FIND "${source}" "    for (i = renderer->bufferContainersToDestroyCount - 1;" buffer_start)
string(FIND "${source}" "    for (i = renderer->textureContainersToDestroyCount - 1;" texture_start)
if(buffer_start LESS 0 OR texture_start LESS 0)
    message(FATAL_ERROR "Pinned SDL Metal retirement anchors changed")
endif()
if(buffer_start LESS texture_start)
    math(EXPR buffer_length "${texture_start} - ${buffer_start}")
    string(SUBSTRING "${source}" ${buffer_start} ${buffer_length} buffer_loop)
    string(SUBSTRING "${source}" ${texture_start} -1 texture_tail)
    string(FIND "${texture_tail}" "    SDL_UnlockMutex(renderer->disposeLock);" texture_length)
    if(texture_length LESS 0)
        message(FATAL_ERROR "Pinned SDL Metal retirement lock boundary changed")
    endif()
    string(SUBSTRING "${texture_tail}" 0 ${texture_length} texture_loop)
    string(REPLACE "${buffer_loop}${texture_loop}" "${texture_loop}${buffer_loop}" source "${source}")
    file(WRITE "${backend}" "${source}")
endif()
string(FIND "${source}" "#include \"gpu_buffer_textures.m\"" buffer_included)
if(buffer_included EQUAL -1)
    file(APPEND "${backend}" "\n#include \"gpu_buffer_textures.m\"\n")
endif()
file(READ "${backend}" source)
string(FIND "${source}" "JULIA_MetalVertexStepRate" vertex_patched)
if(vertex_patched EQUAL -1)
    string(FIND "${source}" "typedef struct MetalRenderer MetalRenderer;" renderer_anchor)
    string(FIND "${source}" "vertexDescriptor.layouts[binding].stepRate = 1;" step_anchor)
    if(renderer_anchor EQUAL -1 OR step_anchor EQUAL -1)
        message(FATAL_ERROR "SDL Metal vertex-input integration anchors changed")
    endif()
    string(REPLACE "typedef struct MetalRenderer MetalRenderer;"
        "typedef struct MetalRenderer MetalRenderer;\n/* Julia extension: full-width per-instance vertex divisors. */\nstatic NSUInteger JULIA_MetalVertexStepRate(SDL_PropertiesID properties, Uint32 slot);" source "${source}")
    string(REPLACE "vertexDescriptor.layouts[binding].stepRate = 1;"
        "vertexDescriptor.layouts[binding].stepRate = JULIA_MetalVertexStepRate(createinfo->props, createinfo->vertex_input_state.vertex_buffer_descriptions[i].slot);" source "${source}")
    file(WRITE "${backend}" "${source}\n#include \"gpu_vertex_input.m\"\n")
endif()
file(READ "${backend}" source)
string(FIND "${source}" "#include \"gpu_texture_views.m\"" views_included)
if(views_included EQUAL -1)
    file(APPEND "${backend}" "\n#include \"gpu_texture_views.m\"\n")
endif()
target_include_directories(SDL3-shared PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

set_target_properties(SDL3-shared PROPERTIES OUTPUT_NAME SDL3Julia)
